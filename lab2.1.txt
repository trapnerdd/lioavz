#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#include <locale.h>

//Функция умножения двух квадратных матриц размером n x n
void multiply_matrices(int n) {
    //Выделение памяти для трех матриц: A, B и результата C
    int** a = (int**)malloc(n * sizeof(int*));
    int** b = (int**)malloc(n * sizeof(int*));
    int** c = (int**)malloc(n * sizeof(int*));

    //Выделение памяти для каждой строки матриц
    for (int i = 0; i < n; i++) {
        a[i] = (int*)malloc(n * sizeof(int));
        b[i] = (int*)malloc(n * sizeof(int));
        c[i] = (int*)malloc(n * sizeof(int));
    }

    //Заполнение матриц A и B случайными числами от 1 до 100
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            a[i][j] = rand() % 100 + 1;
            b[i][j] = rand() % 100 + 1;
        }
    }

    //Умножение матриц: C = A × B
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            c[i][j] = 0;
            for (int k = 0; k < n; k++) {
                c[i][j] += a[i][k] * b[k][j];
            }
        }
    }

}

int main() {
    setlocale(LC_ALL, "");

    //Размеры матриц для тестирования
    int sizes[] = { 100, 200, 400, 1000, 2000, 4000 };
    int num_sizes = 6; //Количество тестируемых размеров

    printf("Размер матрицы | Время в секундах\n");
    printf("------------\n");

    //Тестирование для каждого размера матрицы
    for (int i = 0; i < num_sizes; i++) {
        int n = sizes[i]; //Текущий размер матрицы

        //Включаем таймер
        clock_t start = clock();

        //Умножаем матрицы
        multiply_matrices(n);

        //Останавливаем таймер
        clock_t end = clock();

        //Вычисление времени выполнения в секундах
        double time_taken = ((double)(end - start)) / CLOCKS_PER_SEC;

        //Вывод результатов
        printf("%1d | %1.4f\n", n, time_taken);
    }
    return 0;
}
